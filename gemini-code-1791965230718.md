### 5.2 Capture automatisée de la scène 3D en `.mp4` (`scripts/render-mp4.js`)
Ce script lance Chromium avec accélération WebGL, capture les 85 secondes de l'animation 3D en 1920x1080 et fusionne l'audio avec FFmpeg :

```javascript
const { chromium } = require('playwright');
const path = require('path');
const { execSync } = require('child_process');

(async () => {
  const videoDir = path.join(__dirname, '../dist/raw_video');
  const browser = await chromium.launch({
    headless: false,
    args: ['--use-gl=angle', '--enable-webgl', '--ignore-gpu-blocklist']
  });

  const context = await browser.newContext({
    viewport: { width: 1920, height: 1080 },
    recordVideo: { dir: videoDir, size: { width: 1920, height: 1080 } }
  });

  const page = await context.newPage();
  const fileUrl = 'file://' + path.resolve(__dirname, '../index.html') + '?autoplay=1';
  await page.goto(fileUrl);

  // Attente de 85 secondes (durée exacte du scénario < 1m30)
  await page.waitForTimeout(85500);

  const videoPath = await page.video().path();
  await context.close();
  await browser.close();

  // Fusion Vidéo 3D + Voix-off en .mp4 H.264 conforme au règlement
  const outputMp4 = path.join(__dirname, '../dist/MARTINEZ_Thomas_Grand_Prix_VIE_2026.mp4');
  const voicePath = path.join(__dirname, '../assets/voiceover.mp3');

  execSync(
    `ffmpeg -y -i "${videoPath}" -i "${voicePath}" -c:v libx264 -preset slow -crf 18 -pix_fmt yuv420p -c:a aac -b:a 192k -t 85 "${outputMp4}"`,
    { stdio: 'inherit' }
  );
  console.log('Rendu MP4 final terminé :', outputMp4);
})();
```
